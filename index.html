function detectMaxResolution() { // نفحص 4K بصيغة H.265 (HEVC) أولاً، لأنها الأكثر شيوعاً لتلفزيونات 4K if (canPlayVideo('hvc1.1.6.L153.B0', 3840, 2160, 60) || canPlayVideo('hvc1.1.6.L150.B0', 3840, 2160, 30) || canPlayVideo('hev1.1.6.L153.B0', 3840, 2160, 60)) { return '4K'; } // نفحص 4K بصيغة H.264 (AVC) if (canPlayVideo('avc1.640028', 3840, 2160, 30) || canPlayVideo('avc1.64002A', 3840, 2160, 60) || canPlayVideo('avc1.640033', 3840, 2160, 24)) { return '4K'; } // نفحص 1080p بصيغة H.264 if (canPlayVideo('avc1.640028', 1920, 1080, 60) || canPlayVideo('avc1.4D401F', 1920, 1080, 30)) { return '1080p'; } // إذا فشل كل ما سبق، نرجع 720p كخيار آمن return '720p'; }
